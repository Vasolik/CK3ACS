namespace = acs_evt_save_name

# Name popup for save slot 0. The text is stored on the player under the key acs_save_name_0
acs_evt_save_name.1 = {
	type = character_event
	title = acs_evt_save_name_title
	desc = acs_evt_save_name_desc
	theme = default
	window = acs_save_name_event

	widget = {
		gui = "acs_evt_save_name_enter_text"
		container = "dynamic_birth_name"
		controller = {
			type = text
			data = {
				key = acs_save_name_0
				default = {
					localization_key = acs_evt_save_name_default
					trigger = { always = yes }
				}
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}

	option = {
		name = acs_evt_save_name_save
		acs_se_save_to_slot = { INDEX = 0 }
	}

	option = {
		name = acs_evt_save_name_cancel
		trigger = { always = yes }
	}
}
