# Wraps the generated acs_switch_filter so the "Traits from mods" filter ids (9000, 9001) do not have to live in the generated big switch.
# Scope: the list value scope of acs_vl_filter_active. FILTER : filter id, CANDIDATE : character scope
acs_st_switch_filter_ext = {
	trigger_if = {
		limit = { $FILTER$ = 9000 }
		global_var:acs_gv_main_filters = {
			any_in_list = {
				variable = acs_vl_has_mod_trait
				save_temporary_scope_as = acs_ts_mod_trait_check
				$CANDIDATE$ = { has_trait = scope:acs_ts_mod_trait_check }
			}
		}
	}
	trigger_else_if = {
		limit = { $FILTER$ = 9001 }
		NOT = {
			global_var:acs_gv_main_filters = {
				any_in_list = {
					variable = acs_vl_dont_have_mod_trait
					save_temporary_scope_as = acs_ts_mod_trait_check
					$CANDIDATE$ = { has_trait = scope:acs_ts_mod_trait_check }
				}
			}
		}
	}
	trigger_else_if = {
		limit = {
			$FILTER$ >= 2100
			$FILTER$ <= 2159
		}
		acs_st_switch_resources_filter = { FILTER = $FILTER$ CANDIDATE = $CANDIDATE$ }
	}
	trigger_else = {
		acs_switch_filter = { FILTER = $FILTER$ CANDIDATE = $CANDIDATE$ }
	}
}
